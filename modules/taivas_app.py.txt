import streamlit as st
st.title("V4 TEST 999")
import numpy as np
import matplotlib.pyplot as plt
import streamlit as st

from modules.translations import translations
from modules.country_logic import country_logic
from modules.city_profiles import city_profiles

st.set_page_config(page_title="TAIVAS Energy Control Center", layout="wide")

# ===============================
# Language
# ===============================
language = st.sidebar.selectbox("Language / 語言", ["English","中文"])
t = translations[language]

st.title("TEST VERSION V4")

# ===============================
# Country / City
# ===============================
country_keys = list(country_logic.keys())

country_display_map = {
    t["countries"][k]:k for k in country_keys
}

country_display = st.sidebar.selectbox(
    t["country"],
    list(country_display_map.keys())
)

country_key = country_display_map[country_display]

available_cities = [
    city_key for city_key,city_data in city_profiles.items()
    if city_data["country"] == country_key
]

city_display_map = {
    t["cities"][k]:k for k in available_cities
}

city_display = st.sidebar.selectbox(
    t["city"],
    list(city_display_map.keys())
)

city_key = city_display_map[city_display]

# ===============================
# Energy Inputs
# ===============================

population = st.sidebar.slider(
    t["population"],
    10000,
    5000000,
    500000,
    step=50000
)

solar = st.sidebar.slider(t["solar"],0,500,120,10)
wind = st.sidebar.slider(t["wind"],0,500,80,10)
geothermal = st.sidebar.slider(t["geothermal"],0,500,60,10)
hydro = st.sidebar.slider(t["hydro"],0,500,70,10)
battery = st.sidebar.slider(t["battery"],0,2000,300,50)

# ===============================
# Extreme Climate
# ===============================

scenario = st.sidebar.selectbox(
"Extreme Climate",
[
"Normal",
"Heat Wave",
"Storm",
"Cold Wave",
"Blizzard",
"Typhoon"
]
)

# ===============================
# Storage Failure
# ===============================

failure = st.sidebar.selectbox(
"Storage Failure",
[
"None",
"Partial Failure",
"Major Failure",
"Critical Failure"
]
)

failure_rate = 0

if failure == "Partial Failure":
    failure_rate = 0.3

elif failure == "Major Failure":
    failure_rate = 0.6

elif failure == "Critical Failure":
    failure_rate = 0.9

usable_battery = battery * (1-failure_rate)

# ===============================
# Simulation
# ===============================

hours = np.arange(24)

base_curve = np.array([
0.8,0.75,0.72,0.7,0.7,0.8,
1.0,1.2,1.3,1.4,1.45,1.5,
1.55,1.5,1.4,1.3,1.2,1.1,
1.0,0.9,0.85,0.82,0.8,0.78
])

demand = population/4500 * base_curve

# Climate effects

temp_effect = 0

if scenario == "Heat Wave":
    temp_effect = 0.2

if scenario == "Cold Wave":
    temp_effect = 0.25

demand = demand * (1 + temp_effect)

solar_profile = np.array([
0,0,0,0,0,50,120,220,400,600,720,800,
780,700,500,300,150,50,0,0,0,0,0,0
])/800

if scenario == "Storm" or scenario == "Typhoon":
    solar_profile = solar_profile * 0.3

solar_output = solar * solar_profile

wind_profile = np.array([
0.4,0.45,0.5,0.45,0.4,0.35,0.4,0.45,
0.5,0.55,0.6,0.65,0.6,0.55,0.5,0.45,
0.4,0.35,0.4,0.45,0.5,0.55,0.5,0.45
])

if scenario == "Storm":
    wind_profile = wind_profile * 1.3

wind_output = wind * wind_profile

geo_output = geothermal * 0.9
hydro_output = hydro * 0.85

renewable = solar_output + wind_output + geo_output + hydro_output

battery_level = 0
battery_curve = []
final_supply = []

grid_support = 150

for i in range(24):

    supply = renewable[i]

    diff = supply - demand[i]

    if diff > 0:

        battery_level = min(usable_battery,battery_level+diff)

    else:

        needed = abs(diff)

        use_battery = min(battery_level,needed)

        battery_level -= use_battery

        supply += use_battery

        needed -= use_battery

        supply += min(grid_support,needed)

    battery_curve.append(battery_level)

    final_supply.append(supply)

battery_curve = np.array(battery_curve)
final_supply = np.array(final_supply)

shortfall = np.maximum(demand-final_supply,0).sum()

renewable_ratio = min((renewable.sum()/demand.sum())*100,100)

efficiency = max(0,100-(shortfall/demand.sum()*100))

# ===============================
# Backup Hours
# ===============================

avg_deficit = max(1,np.mean(np.maximum(demand-renewable,0)))

backup_hours = (usable_battery + grid_support) / avg_deficit

# ===============================
# Repair ETA
# ===============================

repair_time = 4

if failure == "Major Failure":
    repair_time = 10

if failure == "Critical Failure":
    repair_time = 24

if scenario == "Typhoon":
    repair_time *= 1.6

if scenario == "Blizzard":
    repair_time *= 1.4

# ===============================
# Water Recovery
# ===============================

humidity = 70

water_recovery = humidity * 15

# ===============================
# KPI
# ===============================

st.subheader("System Performance")

k1,k2,k3,k4,k5,k6 = st.columns(6)

k1.metric("Shortfall",f"{shortfall:.1f} MWh")
k2.metric("Renewables",f"{renewable_ratio:.1f}%")
k3.metric("Efficiency",f"{efficiency:.1f}%")
k4.metric("Backup Hours",f"{backup_hours:.1f}")
k5.metric("Repair ETA",f"{repair_time:.1f} h")
k6.metric("Water Recovery",f"{water_recovery:.0f} L")

# ===============================
# Charts
# ===============================

plt.style.use("dark_background")

col1,col2 = st.columns(2)

with col1:

    fig,ax = plt.subplots()

    ax.plot(hours,demand,label="Demand")

    ax.plot(hours,renewable,label="Renewable")

    ax.plot(hours,final_supply,label="Supply")

    ax.legend()

    ax.set_title("Energy Demand vs Supply")

    st.pyplot(fig)

with col2:

    fig,ax = plt.subplots()

    ax.plot(hours,battery_curve,color="green")

    ax.set_title("Battery Storage")

    st.pyplot(fig)

fig,ax = plt.subplots()

stability = final_supply - demand

colors = ["green" if x>0 else "red" for x in stability]

ax.bar(hours,stability,color=colors)

ax.axhline(0)

ax.set_title("Grid Stability")

st.pyplot(fig)

# ===============================
# Energy Mix
# ===============================

fig,ax = plt.subplots()

values = [
solar_output.sum(),
wind_output.sum(),
geo_output*24,
hydro_output*24
]

labels = [
"Solar",
"Wind",
"Geothermal",
"Hydropower"
]

ax.pie(values,labels=labels,autopct="%1.1f%%")

st.pyplot(fig)

# ===============================
# AI Recommendation
# ===============================

st.subheader("TAIVAS AI Recommendation")

if shortfall < 50:

    st.success("System stable")

else:

    deficit = shortfall

    st.warning("System instability detected")

    st.write(f"Add Solar: {deficit/24:.0f} MW")
    st.write(f"Add Wind: {deficit/28:.0f} MW")
    st.write(f"Add Geothermal: {deficit/30:.0f} MW")
    st.write(f"Increase Battery: {deficit/2:.0f} MWh")

# ===============================
# Recovery analysis
# ===============================

st.subheader("Recovery Analysis")

st.write(f"Backup support: {backup_hours:.1f} hours")

st.write(f"Estimated repair time: {repair_time:.1f} hours")

if repair_time > backup_hours:

    st.error("Repair time exceeds backup support window")

else:

    st.success("System can survive until repair complete")
