import streamlit as st

from modules.translations import translations
from modules.country_logic import country_logic
from modules.city_profiles import city_profiles

st.set_page_config(layout="wide")

# =====================================
# Language
# =====================================
language = st.sidebar.selectbox(
    "Language / 語言",
    list(translations.keys())
)

t = translations[language]

# =====================================
# Safe text helpers
# =====================================
title_text = t.get("title", "TAIVAS Energy Control Center")
subtitle_text = t.get("subtitle", "AI-driven energy resilience and recovery dashboard")
country_label = t.get("country", "Country")
city_label = t.get("city", "City")
population_label = t.get("population", "Population")
solar_label = t.get("solar", "Solar Capacity (MW)")
wind_label = t.get("wind", "Wind Capacity (MW)")
geothermal_label = t.get("geothermal", "Geothermal Capacity (MW)")
hydro_label = t.get("hydro", "Hydropower Capacity (MW)")
battery_label = t.get("battery", "Battery Storage (MWh)")

countries_map = t.get("countries", {})
cities_map = t.get("cities", {})
country_descriptions = t.get("country_descriptions", {})

# =====================================
# Title
# =====================================
st.title(title_text)
st.caption(subtitle_text)

# =====================================
# Country selector
# =====================================
country_keys = list(country_logic.keys())

country_display_map = {
    countries_map.get(k, k): k
    for k in country_keys
}

country_display = st.sidebar.selectbox(
    country_label,
    list(country_display_map.keys())
)

country_key = country_display_map[country_display]

# =====================================
# City selector
# =====================================
available_cities = [
    city_key
    for city_key, city_data in city_profiles.items()
    if city_data.get("country") == country_key
]

city_display_map = {
    cities_map.get(k, k): k
    for k in available_cities
}

city_display = st.sidebar.selectbox(
    city_label,
    list(city_display_map.keys())
)

city_key = city_display_map[city_display]

# =====================================
# Country description
# =====================================
st.subheader(f"{country_label}: {country_display}")

desc = country_descriptions.get(
    country_key,
    country_logic.get(country_key, {}).get("description", f"{country_key} energy model")
)

st.write(desc)

# =====================================
# City profile
# =====================================
st.subheader(f"{city_label}: {city_display}")

city_data = city_profiles[city_key]

st.write(f"Latitude: {city_data.get('lat', '-')}")
st.write(f"Longitude: {city_data.get('lon', '-')}")
st.write(f"Country Model: {city_data.get('country', '-')}")

# =====================================
# Inputs
# =====================================
population = st.sidebar.slider(
    population_label,
    10000,
    5000000,
    500000,
    step=50000
)

solar = st.sidebar.slider(
    solar_label,
    0,
    500,
    120,
    step=10
)

wind = st.sidebar.slider(
    wind_label,
    0,
    500,
    80,
    step=10
)

geothermal = st.sidebar.slider(
    geothermal_label,
    0,
    500,
    60,
    step=10
)

hydro = st.sidebar.slider(
    hydro_label,
    0,
    500,
    70,
    step=10
)

battery = st.sidebar.slider(
    battery_label,
    0,
    2000,
    300,
    step=50
)

# =====================================
# Current inputs
# =====================================
st.subheader("Current Inputs")

st.write(f"{population_label}: {population}")
st.write(f"{solar_label}: {solar}")
st.write(f"{wind_label}: {wind}")
st.write(f"{geothermal_label}: {geothermal}")
st.write(f"{hydro_label}: {hydro}")
st.write(f"{battery_label}: {battery}")
